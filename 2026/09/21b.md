root@gnb:/# cat ~/add_route.sh                                                                                                                                                                   
#!/usr/bin/env bash                                                                                                                                                                              
set -euo pipefail                                                                                                                                                                                
                                                                                                                                                                                                 
if [ -z "${1:-}" ]; then                                                                                                                                                                         
    echo "Usage: $0 <ip-address>"                                                                                                                                                                
    echo "Example: $0 10.189.3.126"                                                                                                                                                              
    exit 1                                                                                                                                                                                       
fi                                                                                                                                                                                               
                                                                                                                                                                                                 
INPUT_IP="$1"                                                                                                                                                                                    
                                                                                                                                                                                                 
# Find interface with IPv4 address starting with 10.189.                                                                                                                                         
IFACE=$(ip -4 -o addr show | awk '                                                                                                                                                               
    $4 ~ /^10\.189\./ {                                                                                                                                                                          
        split($2, iface, "@")                                                                                                                                                                    
        print iface[1]                                                                                                                                                                           
        exit                                                                                                                                                                                     
    }                                                                                                                                                                                            
')                                                                                                                                                                                               
                                                                                                                                                                                                 
if [ -z "${IFACE:-}" ]; then                                                                                                                                                                     
    echo "Error: no interface found with IPv4 address starting with 10.189."                                                                                                                     
    exit 1                                                                                                                                                                                       
fi                                                                                                                                                                                               
                                                                                                                                                                                                 
IFS='.' read -r OCT1 OCT2 OCT3 OCT4 <<< "$INPUT_IP"                                                                                                                                              
                                                                                                                                                                                                 
if [ -z "${OCT1:-}" ] || [ -z "${OCT2:-}" ] || [ -z "${OCT3:-}" ] || [ -z "${OCT4:-}" ]; then                                                                                                    
    echo "Error: invalid IP address: $INPUT_IP"                                                                                                                                                  
    exit 1                                                                                                                                                                                       
fi                                                                                                                                                                                               
                                                                                                                                                                                                 
GATEWAY_IP="${OCT1}.${OCT2}.${OCT3}.$(( (OCT4 / 16) * 16 + 14 ))"                                                                                                                                
                                                                                                                                                                                                 
#echo "Using interface: ${IFACE}"                                                                                                                                                                
echo "Adding route: 10.189.8.0/255.255.255.128 via ${GATEWAY_IP} dev ${IFACE}"                                                                                                                   
                                                                                                                                                                                                 
route add -net 10.189.8.0 netmask 255.255.255.128 gw "$GATEWAY_IP" dev "$IFACE"                                                                                                                  
root@gnb:/#
