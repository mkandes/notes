root@aviat-cli:/# cat setlink.sh                                                                                                                                                                                                                                                                                                                                          
#!/bin/bash                                                                                                                                                                                                                                                                                                                                                               
                                                                                                                                                                                                                                                                                                                                                                          
# Set the path to your CSV file                                                                                                                                                                                                                                                                                                                                           
CSV_FILE="/servers.csv"                                                                                                                                                                                                                                                                                                                                                   
                                                                                                                                                                                                                                                                                                                                                                          
# Check if the CSV file exists                                                                                                                                                                                                                                                                                                                                            
if [ ! -f "$CSV_FILE" ]; then                                                                                                                                                                                                                                                                                                                                             
    echo "Error: CSV file '$CSV_FILE' not found."                                                                                                                                                                                                                                                                                                                         
    exit 1                                                                                                                                                                                                                                                                                                                                                                
fi                                                                                                                                                                                                                                                                                                                                                                        
                                                                                                                                                                                                                                                                                                                                                                          
# Check if a search string was provided as an argument                                                                                                                                                                                                                                                                                                                    
if [ -z "$1" ]; then                                                                                                                                                                                                                                                                                                                                                      
    echo "Usage: source $0 \"<search_string>\""                                                                                                                                                                                                                                                                                                                           
    echo "Note: The script must be sourced to set environment variables."                                                                                                                                                                                                                                                                                                 
    exit 1                                                                                                                                                                                                                                                                                                                                                                
fi                                                                                                                                                                                                                                                                                                                                                                        
                                                                                                                                                                                                                                                                                                                                                                          
# Search for the row that starts with the search string followed by a comma.                                                                                                                                                                                                                                                                                              
# The `^` ensures it matches the beginning of the line (column z).                                                                                                                                                                                                                                                                                                        
# The `-i` flag makes the search case-insensitive.                                                                                                                                                                                                                                                                                                                        
MATCHED_ROW=$(grep -i "^$1," "$CSV_FILE" | head -n 1)                                                                                                                                                                                                                                                                                                                     
                                                                                                                                                                                                                                                                                                                                                                          
# Check if a match was found                                                                                                                                                                                                                                                                                                                                              
if [ -z "$MATCHED_ROW" ]; then                                                                                                                                                                                                                                                                                                                                            
    echo "Error: No matching entry found for '$1' in column z."                                                                                                                                                                                                                                                                                                           
    exit 1                                                                                                                                                                                                                                                                                                                                                                
fi                                                                                                                                                                                                                                                                                                                                                                        
                                                                                                                                                                                                                                                                                                                                                                          
# Use `cut` to extract the values from the a1, a2, and a3 columns.                                                                                                                                                                                                                                                                                                        
# The delimiter is a comma (`,`), and we select fields 2, 3, and 4.                                                                                                                                                                                                                                                                                                       
# `tr` removes carriage returns from Windows-style line endings.                                                                                                                                                                                                                                                                                                          
VALUES=$(echo "$MATCHED_ROW" | tr -d '\r' | cut -d',' -f2,3,4)                                                                                                                                                                                                                                                                                                            
                                                                                                                                                                                                                                                                                                                                                                          
# Use `read` to parse the comma-separated values into variables.                                                                                                                                                                                                                                                                                                          
# The `-r` flag prevents backslash escapes from being interpreted.                                                                                                                                                                                                                                                                                                        
IFS=',' read -r HOST_IP REMOTE_IP API_PORT <<< "$VALUES"                                                                                                                                                                                                                                                                                                                  
                                                                                                                                                                                                                                                                                                                                                                          
# Check if the values were successfully extracted                                                                                                                                                                                                                                                                                                                         
if [ -z "$HOST_IP" ] || [ -z "$REMOTE_IP" ] || [ -z "$API_PORT" ]; then                                                                                                                                                                                                                                                                                                   
    echo "Error: Failed to extract all values from the matched row."                                                                                                                                                                                                                                                                                                      
    exit 1                                                                                                                                                                                                                                                                                                                                                                
fi                                                                                                                                                                                                                                                                                                                                                                        
                                                                                                                                                                                                                                                                                                                                                                          
# Export the variables to make them available in the terminal environment.                                                                                                                                                                                                                                                                                                
export HOST_IP                                                                                                                                                                                                                                                                                                                                                            
export REMOTE_IP                                                                                                                                                                                                                                                                                                                                                          
export API_PORT                                                                                                                                                                                                                                                                                                                                                           
                                                                                                                                                                                                                                                                                                                                                                          
# Print confirmation of the newly set variables.                                                                                                                                                                                                                                                                                                                          
# echo "Environment variables set successfully:"                                                                                                                                                                                                                                                                                                                          
# echo "HOST_IP=$HOST_IP"                                                                                                                                                                                                                                                                                                                                                 
# echo "REMOTE_IP=$REMOTE_IP"
