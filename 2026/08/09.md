import asyncio
import httpx

urls = [
    "https://serverone.com",
    "https://servertwo.com"
]

async def fetch_url_async(client, url):
    try:
        response = await client.get(url, timeout=5.0)
        return {"url": url, "status": response.status_code, "data": response.json()}
    except Exception as e:
        return {"url": url, "error": str(e)}

async def main():
    # Use an async client session for connection pooling
    async with httpx.AsyncClient() as client:
        # Create concurrent tasks for both servers
        tasks = [fetch_url_async(client, url) for url in urls]
        
        # Fire both requests simultaneously and wait for all to finish
        results = await asyncio.gather(*tasks)
        
        for result in results:
            print(result)

# Run the async loop
asyncio.run(main())

```
import concurrent.futures
import requests

# Define your target servers and the payload
urls = ["https://server-one.com", "https://server-two.com"]
payload = {"key": "value"}
headers = {"Content-Type": "application/json"}

# Define a worker function to handle an individual POST request
def send_post(url):
    try:
        # Using a timeout prevents a hanging server from stalling the execution
        response = requests.post(url, json=payload, headers=headers, timeout=5)
        return {"url": url, "status": response.status_code, "data": response.json(), "error": None}
    except Exception as e:
        return {"url": url, "status": None, "data": None, "error": str(e)}

# Send requests simultaneously using a ThreadPoolExecutor
def main():
    # max_workers=2 limits the thread pool to exactly our two servers
    with concurrent.futures.ThreadPoolExecutor(max_workers=2) as executor:
        # map() automatically maps the function to the URL list and executes concurrently
        results = executor.map(send_post, urls)
    
    # Process and print responses
    for result in results:
        print(f"Server: {result['url']}")
        if result['error']:
            print(f"  Error encountered: {result['error']}")
        else:
            print(f"  Status Code: {result['status']}")
            print(f"  Response JSON: {result['data']}")
        print("-" * 40)

if __name__ == "__main__":
    main()
```

import asyncio
import aiohttp

# Define target servers and common payload
urls = ["https://server-one.com", "https://server-two.com"]
payload = {"key": "value"}

async def send_post(session, url):
    try:
        # Define a timeout specifically for this request
        timeout = aiohttp.ClientTimeout(total=5)
        
        # Open the request context
        async with session.post(url, json=payload, timeout=timeout) as response:
            # You must await the JSON parsing method
            data = await response.json()
            return {"url": url, "status": response.status, "data": data, "error": None}
            
    except Exception as e:
        return {"url": url, "status": None, "data": None, "error": str(e)}

async def main():
    # Create a single session to handle connection pooling
    async with aiohttp.ClientSession() as session:
        # Prepare the list of tasks
        tasks = [send_post(session, url) for url in urls]
        
        # Execute both requests simultaneously
        results = await asyncio.gather(*tasks)
    
    # Process results
    for result in results:
        print(f"Server: {result['url']}")
        if result['error']:
            print(f"  Error: {result['error']}")
        else:
            print(f"  Status: {result['status']}")
            print(f"  Response: {result['data']}")
        print("-" * 40)

if __name__ == "__main__":
    asyncio.run(main())

import asyncio
import httpx

# Map each unique server URL to its specific payload
request_data = {
    "https://server-one.com": {"user_id": 101, "action": "login"},
    "https://server-two.com": {"device_id": 999, "status": "active"}
}

async def send_post(client, url, payload):
    try:
        # Pass the specific payload argument into the POST request
        response = await client.post(url, json=payload, timeout=5.0)
        return {"url": url, "status": response.status_code, "data": response.json(), "error": None}
    except Exception as e:
        return {"url": url, "status": None, "data": None, "error": str(e)}

async def main():
    async with httpx.AsyncClient() as client:
        # Unpack items() to pass unique urls and payloads concurrently
        tasks = [send_post(client, url, payload) for url, payload in request_data.items()]
        results = await asyncio.gather(*tasks)
    
    for result in results:
        print(f"Server: {result['url']}")
        if result['error']:
            print(f"  Error: {result['error']}")
        else:
            print(f"  Status: {result['status']}")
            print(f"  Response: {result['data']}")
        print("-" * 40)

if __name__ == "__main__":
    asyncio.run(main())

import asyncio
import aiohttp

request_data = {
    "https://server-one.com": {"user_id": 101, "action": "login"},
    "https://server-two.com": {"device_id": 999, "status": "active"}
}

async def send_post(session, url, payload):
    try:
        timeout = aiohttp.ClientTimeout(total=5)
        # Attach the unique payload to the json parameter
        async with session.post(url, json=payload, timeout=timeout) as response:
            data = await response.json()
            return {"url": url, "status": response.status, "data": data, "error": None}
    except Exception as e:
        return {"url": url, "status": None, "data": None, "error": str(e)}

async def main():
    async with aiohttp.ClientSession() as session:
        tasks = [send_post(session, url, payload) for url, payload in request_data.items()]
        results = await asyncio.gather(*tasks)
    
    for result in results:
        print(f"Server: {result['url']} | Status: {result['status']} | Data: {result['data']}")

if __name__ == "__main__":
    asyncio.run(main())
